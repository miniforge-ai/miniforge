(ns eval.seed.page-fetcher
  (:require [ai.miniforge.schema.interface :as schema]
            [ai.miniforge.store.interface :as store]))

(def ^:private default-page-size
  "Fallback rows-per-page when config is silent. The operational value lives
   in .edn (:page/size); this def is only the silent-config default."
  50)

(defn fetch-page
  "Fetch one page of rows, or pass the store anomaly straight through.
   Page size resolves from config, falling back to default-page-size."
  [config query]
  (let [size   (get config :page/size default-page-size)
        result (store/query query)]
    (if (schema/failed? result)
      result
      (take size (schema/value result)))))
